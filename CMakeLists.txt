cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  if(NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple Silicon architecture")
  endif()
  if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Minimum macOS version")
  endif()
endif()
project(CorridorKeyMac VERSION 0.9.7 LANGUAGES CXX)

set(OPENFX_SDK_ROOT "" CACHE PATH "ASF OpenFX SDK checkout (pinned c3f1c2dcf4f51762b794c4fc21181b4a501be1bf)")
if(NOT EXISTS "${OPENFX_SDK_ROOT}/include/ofxImageEffect.h")
  message(FATAL_ERROR "Set OPENFX_SDK_ROOT to the OpenFX SDK checkout. See scripts/setup_macos.sh.")
endif()
if(NOT UNIX)
  message(FATAL_ERROR "This project requires Unix sockets. The production target is Apple Silicon macOS.")
endif()

find_package(Threads REQUIRED)
add_library(ckmac_core STATIC src/protocol.cpp src/pixels.cpp src/creative.cpp)
target_compile_features(ckmac_core PUBLIC cxx_std_17)
target_include_directories(ckmac_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(ckmac_core PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_libraries(ckmac_core PUBLIC Threads::Threads)

add_library(CorridorKeyMac MODULE src/ofx_plugin.cpp)
target_compile_features(CorridorKeyMac PRIVATE cxx_std_17)
target_include_directories(CorridorKeyMac PRIVATE "${OPENFX_SDK_ROOT}/include")
target_link_libraries(CorridorKeyMac PRIVATE ckmac_core)
set_target_properties(CorridorKeyMac PROPERTIES PREFIX "" SUFFIX ".ofx"
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)

foreach(target ckmac_core CorridorKeyMac)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

if(APPLE)
  enable_language(OBJCXX)
  add_library(ckmac_supervisor STATIC src/worker_manager.mm)
  target_compile_features(ckmac_supervisor PUBLIC cxx_std_17)
  target_include_directories(ckmac_supervisor PUBLIC include)
  target_compile_options(ckmac_supervisor PRIVATE -fobjc-arc)
  target_link_libraries(ckmac_supervisor PUBLIC ckmac_core "-framework Foundation" "-framework Security" "-framework AppKit")
  target_link_libraries(CorridorKeyMac PRIVATE ckmac_supervisor)
  target_compile_definitions(CorridorKeyMac PRIVATE CKMAC_INSTALLED_RUNTIME=1)
  add_executable(CorridorKeyManager MACOSX_BUNDLE src/manager.mm)
  set_target_properties(CorridorKeyManager PROPERTIES OUTPUT_NAME "CorridorKey Mac Manager"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Manager-Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.harvestpictures.CorridorKeyMac.Manager"
    MACOSX_BUNDLE_BUNDLE_VERSION "9071" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.9.7")
  target_compile_options(CorridorKeyManager PRIVATE -fobjc-arc)
  target_link_libraries(CorridorKeyManager PRIVATE ckmac_supervisor "-framework Cocoa")
  add_executable(ckmac-installer-tool src/installer_tool.mm)
  add_executable(ckmac-engine-launcher src/engine_launcher.cpp)
  target_compile_features(ckmac-installer-tool PRIVATE cxx_std_17)
  target_compile_options(ckmac-installer-tool PRIVATE -fobjc-arc -Wno-deprecated-declarations)
  target_link_libraries(ckmac-installer-tool PRIVATE "-framework Foundation")
  set(CK_BUNDLE "${CMAKE_BINARY_DIR}/CorridorKeyMac.ofx.bundle")
  set_target_properties(CorridorKeyMac PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CK_BUNDLE}/Contents/MacOS"
    LIBRARY_OUTPUT_DIRECTORY_DEBUG "${CK_BUNDLE}/Contents/MacOS"
    LIBRARY_OUTPUT_DIRECTORY_RELEASE "${CK_BUNDLE}/Contents/MacOS")
  file(MAKE_DIRECTORY "${CK_BUNDLE}/Contents/Resources")
  configure_file(cmake/Info.plist.in "${CK_BUNDLE}/Contents/Info.plist" @ONLY)
else()
  message(STATUS "Linux build validates portable source only; it is NOT a Mac OFX artifact.")
endif()

include(CTest)

  if(APPLE)
    add_executable(test_installer_transactions src/installer_tool.mm)
    target_compile_definitions(test_installer_transactions PRIVATE CKMAC_INSTALLER_SELF_TEST=1)
    target_compile_features(test_installer_transactions PRIVATE cxx_std_17)
    target_compile_options(test_installer_transactions PRIVATE -fobjc-arc -Wno-deprecated-declarations)
    target_link_libraries(test_installer_transactions PRIVATE "-framework Foundation")
    add_test(NAME installer_transactions COMMAND test_installer_transactions)
  endif()
  add_executable(test_creative tests/test_creative.cpp)
  target_link_libraries(test_creative PRIVATE ckmac_core)
  add_test(NAME creative COMMAND test_creative)
  add_executable(test_pixels tests/test_pixels.cpp)
  target_link_libraries(test_pixels PRIVATE ckmac_core)
  add_test(NAME pixels COMMAND test_pixels)
  add_executable(test_protocol tests/test_protocol.cpp)
  target_link_libraries(test_protocol PRIVATE ckmac_core)
  add_test(NAME protocol COMMAND test_protocol)
  add_executable(test_ofx_host tests/test_ofx_host.cpp src/ofx_plugin.cpp)
  target_include_directories(test_ofx_host PRIVATE "${OPENFX_SDK_ROOT}/include")
  target_link_libraries(test_ofx_host PRIVATE ckmac_core)
  add_test(NAME ofx_host COMMAND test_ofx_host)
  # Invoked by tests/test_cross_language.py with its real Python LocalServer.
  # Not a standalone CTest: it deliberately needs a deterministic fake backend.
  add_executable(bridge_probe tests/bridge_probe.cpp)
  target_link_libraries(bridge_probe PRIVATE ckmac_core)
